\section*{I. Ambiguity (40\%)}

\noindent Paraphrase each of the following sentences to show you understand the ambiguity involved.

\begin{enumerate}
   \item \textbf{Mary finally decided on the boat.}
   \begin{enumerate}
       \item Mary made a final decision while she was on the boat.
       \item Mary made the final decision that she would buy the boat.
   \end{enumerate}

   \item \textbf{The professor's appointment was shocking.}
   \begin{enumerate}
       \item The selection of the professor was shocking.
       \item The scheduling of the professor's meeting was shocking.
   \end{enumerate}

   \item \textbf{I can't recommend him too highly.}
   \begin{enumerate}
       \item I strongly recommend him.
       \item I can't recommend him at all.
   \end{enumerate}

   \item \textbf{John loves his wife and so do I.}
   \begin{enumerate}
       \item John loves his wife, and I also love John's wife.
       \item John loves his wife, and I also have a wife and I love her.
   \end{enumerate}
\end{enumerate}

\section*{II. Constituency Tests (60\%)}

\noindent Apply the constituency tests to the underlined words.

\begin{enumerate}
   \item \textbf{The neighbor's dog ate the groceries that he put \underline{on the table}.}
   \begin{itemize}
       \item Stand Alone Test:
       \begin{itemize}
           \item What did the neighbor's dog eat the groceries that he put?
           \item On the table.
       \end{itemize}
       \item Cleft Sentence Test:
       \begin{itemize}
           \item It is on the table that the neighbor’s dog ate the groceries that he put.
       \end{itemize}
       \item Pro-form Test:
       \begin{itemize}
           \item The neighbor’s dog ate the groceries that he put there.
       \end{itemize}
       \item Topicalization Test:
       \begin{itemize}
           \item On the table, the neighbor’s dog ate the groceries that he put.
       \end{itemize}
       \item Conclusion: \textit{On the table} is a constituent.
   \end{itemize}

   \item \textbf{The neighbor's dog ate \underline{the groceries that he put on the table}.}
   \begin{itemize}
       \item Stand Alone Test:
       \begin{itemize}
           \item What did the neighbor's dog eat?
           \item The groceries that he put on the table.
       \end{itemize}
       \item Cleft Sentence Test:
       \begin{itemize}
           \item It is the groceries that he put on the table that the neighbor’s dog ate.
       \end{itemize}
       \item Pro-form Test:
       \begin{itemize}
           \item The neighbor’s dog ate them.
       \end{itemize}
       \item Topicalization Test:
       \begin{itemize}
           \item The groceries that he put on the table, the neighbor's dog ate.
       \end{itemize}
       \item Conclusion: \textit{The groceries that he put on the table} is a constituent.
   \end{itemize}

   \item \textbf{I saw \underline{Saturn with a telescope}.}
   \begin{itemize}
       \item Stand Alone Test:
       \begin{itemize}
           \item What did you see?
           \item Saturn with a telescope.
       \end{itemize}
       \item Cleft Sentence Test:
       \begin{itemize}
           \item It was Saturn with a telescope that I saw.
       \end{itemize}
       \item Pro-form Test:
       \begin{itemize}
           \item I saw it.
       \end{itemize}
       \item Topicalization Test:
       \begin{itemize}
           \item Saturn with a telescope, I saw.
       \end{itemize}
       \item Conclusion: \textit{Saturn with a telescope} is a constituent.
   \end{itemize}

   \item \textbf{I would never give \underline{a book like that} to my mother as a gift.}
   \begin{itemize}
       \item Stand Alone Test:
       \begin{itemize}
           \item What would you never give to your mother as a gift?
           \item A book like that.
       \end{itemize}
       \item Cleft Sentence Test:
       \begin{itemize}
           \item It is a book like that that I would never give to my mother as a gift.
       \end{itemize}
       \item Pro-form Test:
       \begin{itemize}
           \item I would never give it to my mother as a gift. 
       \end{itemize}
       \item Topicalization Test:
       \begin{itemize}
           \item A book like that, I would never give to my mother as a gift.
       \end{itemize}
       \item Conclusion: \textit{a book like that} is a constituent.
   \end{itemize}
\end{enumerate}
